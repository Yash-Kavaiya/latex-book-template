% Maintainer-owned LaTeX policy.
%
% Author-facing settings all come from book.yml via Pandoc metadata
% variables (title, header-text, ...). Chapter content itself cannot
% inject template structure: raw HTML/TeX is rejected by scripts/validate.py
% before this template ever runs, so only maintainer-reviewed changes to
% this file can alter page layout, header/footer, or packages.
\documentclass[$font-size$$if(class-options)$,$class-options$$endif$]{book}

\usepackage[$paper-size$paper,top=$margin-top$,right=$margin-right$,bottom=$margin-bottom$,left=$margin-left$]{geometry}
% book.cls defaults to \flushbottom, which stretches interline glue to
% justify every page's bottom margin -- on a page with little content
% (e.g. a single figure) that shows up as a large, ugly gap before the
% caption/footer. \raggedbottom lets short pages end early instead.
\raggedbottom

\usepackage{fontspec}
\setmainfont{$main-font$}
\setsansfont{$sans-font$}
\setmonofont{$mono-font$}

\usepackage{xcolor}
$if(link-color-hex)$\definecolor{booklink}{HTML}{$link-color$}$else$\colorlet{booklink}{$link-color$}$endif$
$if(url-color-hex)$\definecolor{bookurl}{HTML}{$url-color$}$else$\colorlet{bookurl}{$url-color$}$endif$
$if(cite-color-hex)$\definecolor{bookcite}{HTML}{$cite-color$}$else$\colorlet{bookcite}{$cite-color$}$endif$
\usepackage[colorlinks=true,linkcolor=booklink,urlcolor=bookurl,citecolor=bookcite,breaklinks=true]{hyperref}
\hypersetup{pdftitle={$title$},pdfauthor={$authors$},pdflang={$lang$}}
\urlstyle{same}

% Math: amsmath enables the align/aligned/cases/... environments Pandoc
% emits for $$...$$ display math; amssymb adds the extended symbol set.
\usepackage{amsmath,amssymb}

% Tables, figures, and page-breaking long tables (see filters/longtable.lua).
\usepackage{graphicx,adjustbox,longtable,booktabs,array,calc,pdflscape}
\setkeys{Gin}{keepaspectratio}
\setlength\LTleft{0pt}
\setlength\LTright{0pt}
\setlength\tabcolsep{4pt}
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\emergencystretch=2em

% Fenced code block syntax highlighting (Shaded/Highlighting environments,
% color definitions); Pandoc fills this in automatically.
$if(highlighting-macros)$
$highlighting-macros$
$endif$

% Running header/footer, driven entirely by book.yml.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\fancyhead[C]{\small $if(header-text)$$header-text$$else$$title$$endif$}
\fancyfoot[C]{\small $footer-text$}
\fancyfoot[R]{\small\thepage}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[R]{\small\thepage}\renewcommand{\headrulewidth}{0pt}}

$if(header-includes)$
$header-includes$
$endif$

\title{$title$}
\author{$authors$}
\date{$date$}

\begin{document}

$if(cover-image)$
\begin{titlepage}
  \centering
  \includegraphics[width=\textwidth,height=\textheight,keepaspectratio]{$cover-image$}
\end{titlepage}
$endif$

\maketitle
$if(subtitle)$
\begin{center}\large $subtitle$\end{center}
$endif$
$if(description)$
\begin{quote}$description$\end{quote}
$endif$

\pagenumbering{$page-numbering$}

$if(table-of-contents)$
\setcounter{tocdepth}{$toc-depth$}
\tableofcontents
\clearpage
$endif$
$if(list-of-figures)$
\listoffigures
\clearpage
$endif$
$if(list-of-tables)$
\listoftables
\clearpage
$endif$

$body$

\end{document}
